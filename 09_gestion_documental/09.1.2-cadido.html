09.1.2

Catálogo de Disposición Documental (CADIDO)

Instrumento que establece los valores documentales (administrativo, legal, fiscal/contable), las vigencias de custodia en Archivo de Trámite y Archivo de Concentración, y el destino final (Baja o Transferencia Histórica) para cada serie del CGCA conforme a la LGA y normatividad CONAC.

Pendiente de desarrollo Análisis funcional canónico (DOC-08) LGA Art. 14 / CFF Art. 30 SIAF-PJEV Transversal
⚖️ Automatización del Ciclo Vital del Documento

El CADIDO es el motor que dota de inteligencia temporal al SIAF-PJEV. Al asociar una serie archivística a este catálogo, el sistema calcula de forma automática la fecha de cierre en trámite y programa la vigencia precautoria en concentración (mínimo 5 años para comprobantes fiscales y contables).

Ningún usuario puede borrar documentos a discreción; cualquier baja física o lógica exige cumplir el plazo del CADIDO y emitir el Acta Formal de Baja aprobada por el AGN y el Consejo Estatal de Archivos.

1 Descripción General

El catálogo 09.1.2 Catálogo de Disposición Documental (CADIDO) complementa al CGCA fijando las reglas de conservación y destino de los documentos del Poder Judicial. Responde a tres preguntas jurídicas indispensables:

  1. ¿Qué valor testimonial tiene la serie? (Administrativo, Legal, Fiscal/Contable o Técnico).
  2. ¿Cuánto tiempo debe conservarse activamente y precautoriamente? (Años en Archivo de Trámite y años en Archivo de Concentración).
  3. ¿Cuál es su destino final una vez extinguidos sus plazos? (Baja Documental autorizada o Transferencia Histórica permanente).

3 Stored Procedures Canónicos (PostgreSQL)
-- SP 1: archivos.fn_leer_cadido
CREATE OR REPLACE FUNCTION archivos.fn_leer_cadido(
    p_filtros JSONB DEFAULT '{}'::jsonb
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_id INT;
    v_cgca VARCHAR(50);
    v_destino VARCHAR(25);
    v_resultado JSONB;
BEGIN
    v_id := (p_filtros->>'cadido_id')::int;
    v_cgca := p_filtros->>'cgca_codigo';
    v_destino := p_filtros->>'destino_final';

    SELECT COALESCE(jsonb_agg(row_to_json(r)), '[]'::jsonb)
    INTO v_resultado
    FROM (
        SELECT 
            d.cadido_id,
            d.cgca_codigo,
            c.seccion,
            c.serie,
            c.subserie,
            d.valor_administrativo,
            d.valor_legal,
            d.valor_fiscal_contable,
            d.valor_tecnico,
            d.vigencia_tramite_anios,
            d.vigencia_concentracion_anios,
            (d.vigencia_tramite_anios + d.vigencia_concentracion_anios) AS vigencia_total_anios,
            d.destino_final,
            d.fundamento_legal,
            d.estatus_registro,
            d.usuario_c,
            d.fecha_c,
            d.usuario_m,
            d.fecha_m
        FROM archivos.cadido_catalogo d
        JOIN archivos.cgca_catalogo c ON d.cgca_codigo = c.cgca_codigo
        WHERE d.estatus_registro = COALESCE((p_filtros->>'estatus_registro')::smallint, 1)
          AND (v_id IS NULL OR d.cadido_id = v_id)
          AND (v_cgca IS NULL OR d.cgca_codigo = v_cgca)
          AND (v_destino IS NULL OR d.destino_final = v_destino)
        ORDER BY c.seccion ASC, c.serie ASC
    ) r;

    RETURN jsonb_build_object('error', 0, 'mensaje', 'OK', 'data', v_resultado);
EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_leer_cadido: ' || SQLERRM, 'data', '[]'::jsonb);
END;
$$;
-- SP 2: archivos.fn_crud_cadido
CREATE OR REPLACE FUNCTION archivos.fn_crud_cadido(
    p_operacion VARCHAR(20),
    p_datos JSONB
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_id INT;
    v_cgca VARCHAR(50);
    v_usuario VARCHAR(50);
BEGIN
    v_usuario := COALESCE(p_datos->>'usuario', 'SISTEMA');

    IF p_operacion = 'INSERT' THEN
        v_cgca := UPPER(TRIM(p_datos->>'cgca_codigo'));

        -- Validar que la serie exista en el CGCA
        IF NOT EXISTS (SELECT 1 FROM archivos.cgca_catalogo WHERE cgca_codigo = v_cgca AND estatus_registro = 1) THEN
            RETURN jsonb_build_object('error', -1, 'mensaje', 'La serie archivística indicada no existe en el CGCA o está inactiva.', 'data', null);
        END IF;

        -- Validar unicidad en CADIDO
        IF EXISTS (SELECT 1 FROM archivos.cadido_catalogo WHERE cgca_codigo = v_cgca AND estatus_registro = 1) THEN
            RETURN jsonb_build_object('error', -1, 'mensaje', 'La serie archivística ya cuenta con disposición documental registrada.', 'data', null);
        END IF;

        INSERT INTO archivos.cadido_catalogo (
            cgca_codigo, valor_administrativo, valor_legal, valor_fiscal_contable,
            valor_tecnico, vigencia_tramite_anios, vigencia_concentracion_anios,
            destino_final, fundamento_legal, estatus_registro, usuario_c, fecha_c
        ) VALUES (
            v_cgca,
            COALESCE((p_datos->>'valor_administrativo')::boolean, FALSE),
            COALESCE((p_datos->>'valor_legal')::boolean, FALSE),
            COALESCE((p_datos->>'valor_fiscal_contable')::boolean, TRUE),
            COALESCE((p_datos->>'valor_tecnico')::boolean, FALSE),
            COALESCE((p_datos->>'vigencia_tramite_anios')::int, 1),
            COALESCE((p_datos->>'vigencia_concentracion_anios')::int, 5),
            COALESCE(p_datos->>'destino_final', 'BAJA_DOCUMENTAL'),
            p_datos->>'fundamento_legal',
            1, v_usuario, clock_timestamp()
        ) RETURNING cadido_id INTO v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Disposición documental asignada con éxito.', 'data', jsonb_build_object('cadido_id', v_id));

    ELSIF p_operacion = 'UPDATE' THEN
        v_id := (p_datos->>'cadido_id')::int;

        UPDATE archivos.cadido_catalogo
        SET valor_administrativo = (p_datos->>'valor_administrativo')::boolean,
            valor_legal = (p_datos->>'valor_legal')::boolean,
            valor_fiscal_contable = (p_datos->>'valor_fiscal_contable')::boolean,
            valor_tecnico = (p_datos->>'valor_tecnico')::boolean,
            vigencia_tramite_anios = (p_datos->>'vigencia_tramite_anios')::int,
            vigencia_concentracion_anios = (p_datos->>'vigencia_concentracion_anios')::int,
            destino_final = p_datos->>'destino_final',
            fundamento_legal = p_datos->>'fundamento_legal',
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE cadido_id = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Disposición documental actualizada exitosamente.', 'data', jsonb_build_object('cadido_id', v_id));

    ELSIF p_operacion = 'DESACTIVAR' THEN
        v_id := (p_datos->>'cadido_id')::int;

        UPDATE archivos.cadido_catalogo
        SET estatus_registro = 0,
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE cadido_id = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Registro CADIDO desactivado.', 'data', jsonb_build_object('cadido_id', v_id));

    ELSE
        RETURN jsonb_build_object('error', -1, 'mensaje', 'Operación no válida: ' || p_operacion, 'data', null);
    END IF;

EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_crud_cadido: ' || SQLERRM, 'data', null);
END;
$$;

Código Nombre Descripción Nivel
