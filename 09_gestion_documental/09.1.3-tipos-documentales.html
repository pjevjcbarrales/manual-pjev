09.1.3

Tipos Documentales y Metadatos Requeridos

Catálogo maestro de tipologías documentales institucionales del SIAF-PJEV (ej. Facturas XML/PDF, Oficios de Ministración, Contratos, Bitácoras, Cédulas de Inventario). Define las extensiones autorizadas, límites de tamaño, obligatoriedad y el esquema de metadatos estandarizados conforme al AGN e ISO 23081.

Pendiente de desarrollo Catálogo Base (DOC-08) ISO 23081 / AGN Transversal SIAF
📋 Estandarización de Requisitos y Metadatos

Este catálogo unifica y gobierna las definiciones dispersas de requisitos documentales del sistema (como 05.1.1 en Cuentas por Pagar y 07.1.5 en Recursos Materiales).

Para cada tipo documental se estipula la extensión permitida (ej. forzosamente .xml o .pdf), el límite de peso en megabytes, si requiere firma electrónica avanzada y cuáles metadatos contextuales deben capturarse u obtenerse automáticamente (ej. UUID SAT, RFC Emisor, No. de Oficio, Monto amparado).

1 Descripción General

El catálogo 09.1.3 Tipos Documentales y Metadatos es el validador funcional del subsistema de archivos. Cuando un usuario intenta cargar un archivo en cualquier formulario del SIAF a través del componente <EntityFiles /> o <SingleFileField />, el sistema consulta este catálogo para aplicar las restricciones técnicas y registrar los metadatos correspondientes:

3 Stored Procedures Canónicos
-- SP 1: archivos.fn_leer_tipo_documento
CREATE OR REPLACE FUNCTION archivos.fn_leer_tipo_documento(
    p_filtros JSONB DEFAULT '{}'::jsonb
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_id INT;
    v_clave VARCHAR(50);
    v_resultado JSONB;
BEGIN
    v_id := (p_filtros->>'id_tipo_documento')::int;
    v_clave := p_filtros->>'clave_tipo';

    SELECT COALESCE(jsonb_agg(row_to_json(r)), '[]'::jsonb)
    INTO v_resultado
    FROM (
        SELECT 
            t.id_tipo_documento,
            t.clave_tipo,
            t.nombre_tipo,
            t.extensiones_permitidas,
            t.tamano_max_mb,
            t.requiere_firma_electronica,
            t.esquema_metadatos_json,
            t.estatus_registro,
            t.usuario_c,
            t.fecha_c,
            t.usuario_m,
            t.fecha_m
        FROM archivos.tipo_documento_catalogo t
        WHERE t.estatus_registro = COALESCE((p_filtros->>'estatus_registro')::smallint, 1)
          AND (v_id IS NULL OR t.id_tipo_documento = v_id)
          AND (v_clave IS NULL OR t.clave_tipo ILIKE '%' || v_clave || '%')
        ORDER BY t.nombre_tipo ASC
    ) r;

    RETURN jsonb_build_object('error', 0, 'mensaje', 'OK', 'data', v_resultado);
EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_leer_tipo_documento: ' || SQLERRM, 'data', '[]'::jsonb);
END;
$$;
-- SP 2: archivos.fn_crud_tipo_documento
CREATE OR REPLACE FUNCTION archivos.fn_crud_tipo_documento(
    p_operacion VARCHAR(20),
    p_datos JSONB
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_id INT;
    v_clave VARCHAR(50);
    v_usuario VARCHAR(50);
BEGIN
    v_usuario := COALESCE(p_datos->>'usuario', 'SISTEMA');

    IF p_operacion = 'INSERT' THEN
        v_clave := UPPER(TRIM(p_datos->>'clave_tipo'));

        IF EXISTS (SELECT 1 FROM archivos.tipo_documento_catalogo WHERE clave_tipo = v_clave) THEN
            RETURN jsonb_build_object('error', -1, 'mensaje', 'La clave de tipo documental ya existe: ' || v_clave, 'data', null);
        END IF;

        INSERT INTO archivos.tipo_documento_catalogo (
            clave_tipo, nombre_tipo, extensiones_permitidas, mime_permitidos,
            tamano_max_mb, requiere_firma_electronica, esquema_metadatos_json,
            estatus_registro, usuario_c, fecha_c
        ) VALUES (
            v_clave,
            p_datos->>'nombre_tipo',
            LOWER(p_datos->>'extensiones_permitidas'),
            p_datos->>'mime_permitidos',
            COALESCE((p_datos->>'tamano_max_mb')::numeric, 15.0),
            COALESCE((p_datos->>'requiere_firma_electronica')::boolean, FALSE),
            p_datos->'esquema_metadatos_json',
            1, v_usuario, clock_timestamp()
        ) RETURNING id_tipo_documento INTO v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Tipo documental creado exitosamente.', 'data', jsonb_build_object('id_tipo_documento', v_id));

    ELSIF p_operacion = 'UPDATE' THEN
        v_id := (p_datos->>'id_tipo_documento')::int;

        UPDATE archivos.tipo_documento_catalogo
        SET nombre_tipo = p_datos->>'nombre_tipo',
            extensiones_permitidas = LOWER(p_datos->>'extensiones_permitidas'),
            mime_permitidos = p_datos->>'mime_permitidos',
            tamano_max_mb = (p_datos->>'tamano_max_mb')::numeric,
            requiere_firma_electronica = (p_datos->>'requiere_firma_electronica')::boolean,
            esquema_metadatos_json = p_datos->'esquema_metadatos_json',
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE id_tipo_documento = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Tipo documental actualizado con éxito.', 'data', jsonb_build_object('id_tipo_documento', v_id));

    ELSIF p_operacion = 'DESACTIVAR' THEN
        v_id := (p_datos->>'id_tipo_documento')::int;

        UPDATE archivos.tipo_documento_catalogo
        SET estatus_registro = 0,
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE id_tipo_documento = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Tipo documental desactivado.', 'data', jsonb_build_object('id_tipo_documento', v_id));

    ELSE
        RETURN jsonb_build_object('error', -1, 'mensaje', 'Operación no válida: ' || p_operacion, 'data', null);
    END IF;

EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_crud_tipo_documento: ' || SQLERRM, 'data', null);
END;
$$;
