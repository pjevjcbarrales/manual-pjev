09.1.4

Catálogo de Sellos Digitales Institucionales

Catálogo de configuración de sellos visuales y criptográficos del Poder Judicial (modelo DocuWare Stamps). Define la apariencia gráfica, los roles facultados, las exigencias de e.firma y los disparadores de transición transaccional en el ciclo de vida del gasto o ingreso.

Pendiente de desarrollo Configuración Workflow (DOC-08) Benchmark: DocuWare Stamps
🖋️ Sellos con Fuerza Probatoria y Triggers de Estado

A diferencia de un simple "check" en un formulario, los Sellos Digitales operan como en DocuWare: imprimen un distintivo visual estilizado en el visor sobre el documento (con nombre del funcionario, cargo, fecha, hora y folio) y al mismo tiempo asientan un registro criptográfico inmutable en la bitácora (Art. 45 LGA).

Asimismo, el catálogo permite configurar si el sello detona la transición del trámite padre (ej. estampar "AUTORIZADO PARA PAGO" cambia automáticamente la Afectación de Gasto a estado AUTORIZADA y habilita la generación de la Orden de Pago).

3 Stored Procedures Canónicos
-- SP 1: archivos.fn_leer_catalogo_sellos
CREATE OR REPLACE FUNCTION archivos.fn_leer_catalogo_sellos(
    p_filtros JSONB DEFAULT '{}'::jsonb
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_resultado JSONB;
BEGIN
    SELECT COALESCE(jsonb_agg(row_to_json(r)), '[]'::jsonb)
    INTO v_resultado
    FROM (
        SELECT 
            s.id_sello,
            s.clave_sello,
            s.nombre_sello,
            s.color_hex,
            s.roles_permitidos,
            s.requiere_efirma,
            s.estatus_disparado,
            s.estatus_registro,
            s.usuario_c,
            s.fecha_c
        FROM archivos.sello_catalogo s
        WHERE s.estatus_registro = COALESCE((p_filtros->>'estatus_registro')::smallint, 1)
        ORDER BY s.id_sello ASC
    ) r;

    RETURN jsonb_build_object('error', 0, 'mensaje', 'OK', 'data', v_resultado);
EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_leer_catalogo_sellos: ' || SQLERRM, 'data', '[]'::jsonb);
END;
$$;
-- SP 2: archivos.fn_crud_catalogo_sellos
CREATE OR REPLACE FUNCTION archivos.fn_crud_catalogo_sellos(
    p_operacion VARCHAR(20),
    p_datos JSONB
)
RETURNS JSONB
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = archivos, public
AS $$
DECLARE
    v_id INT;
    v_clave VARCHAR(50);
    v_usuario VARCHAR(50);
BEGIN
    v_usuario := COALESCE(p_datos->>'usuario', 'SISTEMA');

    IF p_operacion = 'INSERT' THEN
        v_clave := UPPER(TRIM(p_datos->>'clave_sello'));

        IF EXISTS (SELECT 1 FROM archivos.sello_catalogo WHERE clave_sello = v_clave) THEN
            RETURN jsonb_build_object('error', -1, 'mensaje', 'La clave de sello ya existe: ' || v_clave, 'data', null);
        END IF;

        INSERT INTO archivos.sello_catalogo (
            clave_sello, nombre_sello, color_hex, roles_permitidos,
            requiere_efirma, estatus_disparado, estatus_registro, usuario_c, fecha_c
        ) VALUES (
            v_clave,
            p_datos->>'nombre_sello',
            COALESCE(p_datos->>'color_hex', '#059669'),
            p_datos->>'roles_permitidos',
            COALESCE((p_datos->>'requiere_efirma')::boolean, FALSE),
            p_datos->>'estatus_disparado',
            1, v_usuario, clock_timestamp()
        ) RETURNING id_sello INTO v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Sello institucional registrado con éxito.', 'data', jsonb_build_object('id_sello', v_id));

    ELSIF p_operacion = 'UPDATE' THEN
        v_id := (p_datos->>'id_sello')::int;

        UPDATE archivos.sello_catalogo
        SET nombre_sello = p_datos->>'nombre_sello',
            color_hex = p_datos->>'color_hex',
            roles_permitidos = p_datos->>'roles_permitidos',
            requiere_efirma = (p_datos->>'requiere_efirma')::boolean,
            estatus_disparado = p_datos->>'estatus_disparado',
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE id_sello = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Sello institucional actualizado.', 'data', jsonb_build_object('id_sello', v_id));

    ELSIF p_operacion = 'DESACTIVAR' THEN
        v_id := (p_datos->>'id_sello')::int;

        UPDATE archivos.sello_catalogo
        SET estatus_registro = 0,
            usuario_m = v_usuario,
            fecha_m = clock_timestamp()
        WHERE id_sello = v_id;

        RETURN jsonb_build_object('error', 0, 'mensaje', 'Sello institucional desactivado.', 'data', jsonb_build_object('id_sello', v_id));

    ELSE
        RETURN jsonb_build_object('error', -1, 'mensaje', 'Operación no válida: ' || p_operacion, 'data', null);
    END IF;

EXCEPTION WHEN OTHERS THEN
    RETURN jsonb_build_object('error', -2, 'mensaje', 'Error en fn_crud_catalogo_sellos: ' || SQLERRM, 'data', null);
END;
$$;
